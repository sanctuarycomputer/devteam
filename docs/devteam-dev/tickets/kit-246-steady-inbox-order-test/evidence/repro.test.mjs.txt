// kit-246 repro: the goal drawer's "Wrong?" press, with the goal's record (its /api/peek) landing
// between p.click's measure and its press. REPRO_VARIANT=race measures first (main's order);
// REPRO_VARIANT=fixed waits for the record to be drawn, then measures (the fixed test's order).
// Copied into test/ by repro.sh so its ./lib imports resolve; never committed there.
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { canDrive, openChrome } from './lib/chrome.mjs';
import { freePort, json, PORTS, startDashboard, until } from './lib/kit.mjs';
import { fixture } from './lib/next-step-fixture.mjs';

const VARIANT = process.env.REPRO_VARIANT || 'race';

test(`repro (${VARIANT}): Wrong? measured, then the goal's record lands, then the press`, { skip: canDrive ? false : 'needs Chrome' }, async () => {
  const F = fixture('nbs-repro');
  const { old, ok, goal, impact } = F;
  const G = goal('Team billing', 1);
  const D = impact(G, 3, 'W2 is revenue.');
  const port = await freePort();
  const dash = startDashboard(old, F.env({ DEVTEAM_HUB_PORT: String(PORTS + 10), DEVTEAM_PORT_BASE: String(PORTS + 11) }), ['--port', String(port)]);
  const chrome = await openChrome();
  const p = await chrome.newPage();
  try {
    await until(() => /dashboard on http/.test(dash.out()), 'the dashboard');
    await p.send('Page.enable');
    await p.send('Runtime.enable');
    await p.send('Page.navigate', { url: `http://127.0.0.1:${port}/#questions` });
    const soon = (expr, what) => until(() => p.ev(expr), what);
    await soon(`!!document.querySelector('#questions')?.children.length`, 'the Inbox');
    // hold every /api/peek answer until released: the drawer's record arrives when the test says
    await p.ev(`void (() => { window.__hold = new Promise((r) => { window.__release = r; }); const f = window.fetch; window.fetch = async (u, o) => { if (String(u).startsWith('/api/peek')) await window.__hold; return f(u, o); }; })()`);
    await p.ev(`void (location.hash = '#goal/${G}')`);
    const key = `imp:${D}`;
    const btn = `[data-correct="${key}"]`;
    await soon(`!!document.querySelector('${btn}')`, 'the goal drawer');
    // kit-88's slide-in, on a dashboard that has one: measured only once the drawer is at rest
    await soon(`(() => { const d = document.getElementById('drawer'); return getComputedStyle(d).transform === 'none' && !d.getAnimations().some((a) => a.transitionProperty === 'transform'); })()`, 'the goal drawer at rest');
    // the record drawn: its Created row there (before and after kit-88) and no Loading… left
    const loaded = `[...document.querySelectorAll('#drawer-body dt')].some((dt) => dt.textContent.trim().startsWith('Created')) && !document.querySelector('#drawer-body .loading')`;
    console.log(`REPRO drawer drawn; record loaded: ${await p.ev(loaded)}`);
    if (VARIANT === 'fixed') { await p.ev('window.__release()'); await soon(loaded, 'the goal\'s record'); }
    const at = await p.at(btn); // what p.click measures
    console.log(`REPRO Wrong? measured at y=${Math.round(at.y)}`);
    if (VARIANT === 'race') { await p.ev('window.__release()'); await soon(loaded, 'the goal\'s record'); }
    console.log(`REPRO at press time Wrong? is at y=${Math.round(await p.ev(`(() => { const r = document.querySelector('${btn}').getBoundingClientRect(); return r.top + r.height / 2; })()`))}; the point is on ${await p.ev(`(() => { const e = document.elementFromPoint(${at.x}, ${at.y}); return e ? e.tagName + ' ' + JSON.stringify(e.textContent.trim().slice(0, 40)) : 'nothing'; })()`)}`);
    await p.clickAt(at.x, at.y); // what p.click presses
    console.log(`REPRO after the press the form is ${await p.ev(`document.querySelector('[data-correct-form="${key}"]').hidden ? 'hidden' : 'open'`)}`);
    await p.ev(`(() => { document.querySelector('select[data-impact="${key}"]').value = '4'; document.querySelector('input[data-reason="${key}"]').value = 'the launch blocker'; })()`);
    await p.click(`[data-correct-send="${key}"]`);
    await until(() => json(ok(old, ['goal', 'show', G])).impact.level === 4, 'the owner\'s value');
    console.log('REPRO the owner\'s value landed');
  } finally {
    p.close();
    await chrome.close();
    await dash.stop();
    F.done();
  }
});
